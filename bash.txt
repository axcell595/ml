import pandas as pd
from datetime import datetime
from openpyxl import Workbook


# Load the exported CSV from Xero
input_file = 'invoices.csv'  # Change if your file name is different
df = pd.read_csv(input_file)

# Convert DueDate to datetime (adjust column name if needed)
df['DueDate'] = pd.to_datetime(df['DueDate'])

# Today's date (automatically uses current date when run)
today = datetime.now().date()

# Filter for outstanding invoices (adjust 'Status' value if your Xero export uses different terms)
df = df[(df['InvoiceAmountDue'] > 0) & (df['Status'] == 'Awaiting Payment')]

# Compute custom Status: 'Overdue' if past due, else 'Due'
df['Status'] = df['DueDate'].apply(lambda d: 'Overdue' if d.date() < today else 'Due')

# Select relevant columns including Currency for filtering
temp_columns = ['ContactName', 'InvoiceNumber', 'Status', 'DueDate', 'Total', 'InvoiceAmountPaid', 'InvoiceAmountDue', 'Currency']
df = df[temp_columns]

# Separate into EUR and USD, dropping Currency
eur_df = df[df['Currency'] == 'EUR'].drop('Currency', axis=1)
usd_df = df[df['Currency'] == 'USD'].drop('Currency', axis=1)

# Rename columns to match your desired output
rename_dict = {
    'ContactName': 'Customer Name',
    'InvoiceNumber': 'Invoice #',
    'Status': 'Payment Status',
    'DueDate': 'Due date',
    'Total': 'Total Amount',
    'InvoiceAmountPaid': 'Amount Paid',
    'InvoiceAmountDue': 'Balance'
}
eur_df = eur_df.rename(columns=rename_dict)
usd_df = usd_df.rename(columns=rename_dict)

# Format Due date as YYYY-MM-DD (no time)
eur_df['Due date'] = eur_df['Due date'].dt.strftime('%d-%m-%Y')
usd_df['Due date'] = usd_df['Due date'].dt.strftime('%d-%m-%Y')

# Final columns order
final_columns = ['Customer Name', 'Invoice #', 'Status', 'Due date', 'Total Amount', 'Amount Paid', 'Balance']

# Add total outstanding row for EUR
eur_sum = eur_df['Balance'].sum()
eur_total = pd.DataFrame([['Total Outstanding', '', '', '', '', '', eur_sum]], columns=final_columns)
eur_df = pd.concat([eur_df, eur_total], ignore_index=True)

# Add total outstanding row for USD
usd_sum = usd_df['Balance'].sum()
usd_total = pd.DataFrame([['Total Outstanding', '', '', '', '', '', usd_sum]], columns=final_columns)
usd_df = pd.concat([usd_df, usd_total], ignore_index=True)

# Write to Excel file with separate sheets
output_file = 'fo_out.xlsx'
with pd.ExcelWriter(output_file) as writer:
    eur_df.to_excel(writer, sheet_name='EUR', index=False)
    usd_df.to_excel(writer, sheet_name='USD', index=False)

print(f'Processing complete! Output saved to {output_file}')
